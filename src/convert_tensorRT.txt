import tensorrt as trt

def build_engine(onnx_path, engine_path):
    # 1. Khởi tạo Logger
    logger = trt.Logger(trt.Logger.INFO)
    
    # 2. Tạo Builder và Network
    builder = trt.Builder(logger)
    network = builder.create_network(1 << int(trt.NetworkDefinitionCreationFlag.EXPLICIT_BATCH))
    parser = trt.OnnxParser(network, logger)
    
    # 3. Đọc file ONNX
    with open(onnx_path, 'rb') as f:
        if not parser.parse(f.read()):
            print("ERROR: Failed to parse the ONNX file.")
            for error in range(parser.num_errors):
                print(parser.get_error(error))
            return None

    # 4. Tạo Config và thiết lập thông số
    config = builder.create_builder_config()
    
    # Set Workspace (4GB như bạn muốn)
    config.set_memory_pool_limit(trt.MemoryPoolType.WORKSPACE, 4 * 1024 * 1024 * 1024)
    
    # Bật FP16 (Cái này cực kỳ nên bật cho 2080 Ti nhé!)
    if builder.platform_has_fast_fp16:
        config.set_flag(trt.BuilderFlag.FP16)

    # 5. Thiết lập Dynamic Shapes (Min/Opt/Max)
    profile = builder.create_optimization_profile()
    input_name = network.get_input(0).name
    profile.set_shape(input_name, (1, 3, 320, 320), (8, 3, 640, 640), (32, 3, 640, 640))
    config.add_optimization_profile(profile)

    # 6. Tiến hành Build Engine
    print(f"Building TensorRT engine: {engine_path}...")
    engine_bytes = builder.build_serialized_network(network, config)
    
    # 7. Lưu file .plan
    with open(engine_path, "wb") as f:
        f.write(engine_bytes)
    print("Build hoàn tất!")

# Chạy thử
build_engine("yolov8n.onnx", "model.plan")
